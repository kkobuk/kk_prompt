import json
import numpy as np
from casadi import SX, vertcat, Function, nlpsol

# 로컬 파일 로딩
with open("sensor_data.json", "r") as f:
    sensor_data = json.load(f)

# 예: LLM이 생성한 제어 목표 (로컬 LLM 활용 예시)
llm_generated_target = {"Temperature": 75.0, "Pressure": 2.0}  # 목표 상태

# 시스템 모델 정의 (단순 1차 시스템 예)
x = SX.sym('x', 2)  # [Temperature, Pressure]
u = SX.sym('u', 2)  # [HeaterPower, ValveOpening]

# 상태 방정식 예시
x_next = x + vertcat(0.1*(u[0] - x[0]), 0.05*(u[1] - x[1]))

f = Function('f', [x, u], [x_next])

# MPC 최적화 문제 정의
N = 10  # 예측 horizon
U = SX.sym('U', 2*N)
X = [x]
for k in range(N):
    X.append(f(X[-1], U[2*k:2*k+2]))

# 비용 함수 (목표 상태와 오차)
cost = 0
for k in range(1, N+1):
    cost += (X[k][0] - llm_generated_target["Temperature"])**2
    cost += (X[k][1] - llm_generated_target["Pressure"])**2

nlp = {'x': U, 'f': cost}
solver = nlpsol('solver', 'ipopt', nlp)
res = solver(x0=np.zeros(2*N))
optimal_u = res['x'].full().flatten()[:2]
print("Optimal control action:", optimal_u)

# 로컬 LLM은 HuggingFace transformers 활용 예시

from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

class LocalLLM:
    def __init__(self, model_name="TheBloke/LLaMA-3B-GPTQ"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForCausalLM.from_pretrained(model_name, device_map="auto", trust_remote_code=True)

    def generate_target(self, prompt):
        inputs = self.tokenizer(prompt, return_tensors="pt").to("cuda")
        output = self.model.generate(**inputs, max_new_tokens=50)
        text = self.tokenizer.decode(output[0], skip_special_tokens=True)
        # 간단 파싱 (JSON 형태로 반환 가정)
        import json
        try:
            return json.loads(text)
        except:
            return {"Temperature": 70.0, "Pressure": 2.0}  # 기본값

# mpc_controller.py

import numpy as np
from casadi import SX, vertcat, Function, nlpsol

class MPCController:
    def __init__(self, horizon=10):
        self.N = horizon  # 예측 horizon

    def solve(self, x0, target):
        # 상태, 제어 변수 정의
        x = SX.sym('x', 2)  # [Temperature, Pressure]
        u = SX.sym('u', 2)  # [HeaterPower, ValveOpening]

        # 단순 시스템 모델
        x_next = x + vertcat(0.1*(u[0] - x[0]), 0.05*(u[1] - x[1]))
        f = Function('f', [x, u], [x_next])

        # 최적화 변수
        U = SX.sym('U', 2*self.N)
        X = [x]
        for k in range(self.N):
            X.append(f(X[-1], U[2*k:2*k+2]))

        # 비용 함수
        cost = 0
        for k in range(1, self.N+1):
            cost += (X[k][0] - target["Temperature"])**2
            cost += (X[k][1] - target["Pressure"])**2

        nlp = {'x': U, 'f': cost}
        solver = nlpsol('solver', 'ipopt', nlp)
        res = solver(x0=np.zeros(2*self.N))
        optimal_u = res['x'].full().flatten()[:2]
        return optimal_u

2️⃣ MPC 제약 조건 자동 생성 (mpc_controller.py)
import numpy as np
from casadi import SX, vertcat, Function, nlpsol

class MPCController:
    def __init__(self, horizon=10, state_vars=None, control_vars=None):
        self.N = horizon
        self.state_vars = state_vars  # 온톨로지 기반 상태 변수
        self.control_vars = control_vars  # 온톨로지 기반 제어 변수

    def solve(self, x0, target):
        x = SX.sym('x', len(self.state_vars))
        u = SX.sym('u', len(self.control_vars))

        # 단순 선형 모델 예제
        x_next = x + vertcat(
            0.1*(u[0] - x[0]),
            0.05*(u[1] - x[1])
        )
        f = Function('f', [x, u], [x_next])

        # 최적화 변수
        U = SX.sym('U', len(self.control_vars)*self.N)
        X = [x]
        for k in range(self.N):
            X.append(f(X[-1], U[len(self.control_vars)*k:len(self.control_vars)*(k+1)]))

        # 비용 함수 (목표 상태 오차)
        cost = 0
        for k in range(1, self.N+1):
            for i, state in enumerate(self.state_vars):
                cost += (X[k][i] - target[state.name])**2

        # 제약 조건 자동 생성
        g = []
        for k in range(self.N):
            u_k = U[len(self.control_vars)*k:len(self.control_vars)*(k+1)]
            # 제어 변수 범위
            for i, cv in enumerate(self.control_vars):
                g.append(u_k[i] - cv.max_val)  # u <= max
                g.append(cv.min_val - u_k[i])  # u >= min
            # 상태 변수 범위
            for i, sv in enumerate(self.state_vars):
                g.append(X[k+1][i] - sv.max_val)  # x <= max
                g.append(sv.min_val - X[k+1][i])  # x >= min

        nlp = {'x': U, 'f': cost, 'g': vertcat(*g)}
        solver = nlpsol('solver', 'ipopt', nlp)
        res = solver(x0=np.zeros(len(self.control_vars)*self.N))
        optimal_u = res['x'].full().flatten()[:len(self.control_vars)]
        return optimal_u

## 1️⃣ `main_sim_plot.py` (실시간 플롯 버전)

```python id="v9p3h4"
import numpy as np
import matplotlib.pyplot as plt
from ontology import *
from llm_interface import LocalLLM
from mpc_controller import MPCController

# 시뮬레이션 초기 상태
x = np.array([65.0, 1.5])  # Temperature, Pressure 초기값

# 로컬 LLM로 목표 상태 생성
llm = LocalLLM()
prompt = "Set target state for the system based on current sensor data."
target_state = llm.generate_target(prompt)
print("Target state:", target_state)

# MPC 컨트롤러 초기화
mpc = MPCController(
    horizon=10,
    state_vars=[temperature, pressure],
    control_vars=[heater_power, valve_opening]
)

# 시뮬레이션 루프
time_steps = 20
history = []

# 실시간 플롯 설정
plt.ion()
fig, axs = plt.subplots(2, 1, figsize=(10, 6))
axs[0].set_title("States over Time")
axs[0].set_ylabel("Value")
axs[1].set_title("Control Actions over Time")
axs[1].set_ylabel("Control Value")
axs[1].set_xlabel("Time Step")

for t in range(time_steps):
    # 1. 현재 상태에서 MPC 계산
    u_opt = mpc.solve(x, target_state)
    
    # 2. 상태 업데이트
    x = x + np.array([0.1*(u_opt[0] - x[0]), 0.05*(u_opt[1] - x[1])])
    
    # 3. 히스토리 기록
    history.append({"time": t, "Temperature": x[0], "Pressure": x[1],
                    "HeaterPower": u_opt[0], "ValveOpening": u_opt[1]})
    
    # 4. 실시간 플롯 업데이트
    axs[0].cla()
    axs[1].cla()
    
    axs[0].plot([h["time"] for h in history], [h["Temperature"] for h in history], label="Temperature")
    axs[0].plot([h["time"] for h in history], [h["Pressure"] for h in history], label="Pressure")
    axs[0].set_title("States over Time")
    axs[0].set_ylabel("Value")
    axs[0].legend()
    
    axs[1].plot([h["time"] for h in history], [h["HeaterPower"] for h in history], label="HeaterPower")
    axs[1].plot([h["time"] for h in history], [h["ValveOpening"] for h in history], label="ValveOpening")
    axs[1].set_title("Control Actions over Time")
    axs[1].set_ylabel("Control Value")
    axs[1].set_xlabel("Time Step")
    axs[1].legend()
    
    plt.pause(0.5)  # 반초 간격으로 업데이트

plt.ioff()
plt.show()

# 최종 히스토리 저장
import json
with open("simulation_history.json", "w") as f:
    json.dump(history, f, indent=2)

print("Simulation complete. History saved to simulation_history.json")
```

3. 로컬 LLM 인터페이스
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

class LocalLLM:
    def __init__(self, model_name="TheBloke/LLaMA-3B-GPTQ"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForCausalLM.from_pretrained(
            model_name, device_map="auto", trust_remote_code=True
        )

    def generate_target(self, prompt):
        inputs = self.tokenizer(prompt, return_tensors="pt").to("cuda")
        output = self.model.generate(**inputs, max_new_tokens=50)
        text = self.tokenizer.decode(output[0], skip_special_tokens=True)
        import json
        try:
            return json.loads(text)
        except:
            return {"Temperature": 70.0, "Pressure": 2.0}  # 기본값
4. MPC 컨트롤러 (온톨로지 기반 제약 조건 포함)
import numpy as np
from casadi import SX, vertcat, Function, nlpsol

class MPCController:
    def __init__(self, horizon=10, state_vars=None, control_vars=None):
        self.N = horizon
        self.state_vars = state_vars
        self.control_vars = control_vars

    def solve(self, x0, target):
        x = SX.sym('x', len(self.state_vars))
        u = SX.sym('u', len(self.control_vars))

        # 시스템 모델 예제
        x_next = x + vertcat(0.1*(u[0]-x[0]), 0.05*(u[1]-x[1]))
        f = Function('f', [x, u], [x_next])

        # 최적화 변수
        U = SX.sym('U', len(self.control_vars)*self.N)
        X = [x]
        for k in range(self.N):
            X.append(f(X[-1], U[len(self.control_vars)*k:len(self.control_vars)*(k+1)]))

        # 비용 함수
        cost = 0
        for k in range(1, self.N+1):
            for i, state in enumerate(self.state_vars):
                cost += (X[k][i] - target[state.name])**2

        # 제약 조건 자동 생성
        g = []
        for k in range(self.N):
            u_k = U[len(self.control_vars)*k:len(self.control_vars)*(k+1)]
            for i, cv in enumerate(self.control_vars):
                g.append(u_k[i] - cv.max_val)
                g.append(cv.min_val - u_k[i])
            for i, sv in enumerate(self.state_vars):
                g.append(X[k+1][i] - sv.max_val)
                g.append(sv.min_val - X[k+1][i])

        nlp = {'x': U, 'f': cost, 'g': vertcat(*g)}
        solver = nlpsol('solver', 'ipopt', nlp)
        res = solver(x0=np.zeros(len(self.control_vars)*self.N))
        return res['x'].full().flatten()[:len(self.control_vars)]

import numpy as np
import matplotlib.pyplot as plt
from ontology import *
from llm_interface import LocalLLM
from mpc_controller import MPCController

x = np.array([65.0, 1.5])
llm = LocalLLM()
prompt = "Set target state for the system based on current sensor data."
target_state = llm.generate_target(prompt)
print("Target state:", target_state)

mpc = MPCController(horizon=10, state_vars=[temperature, pressure],
                    control_vars=[heater_power, valve_opening])

time_steps = 20
history = []

plt.ion()
fig, axs = plt.subplots(2, 1, figsize=(10,6))

for t in range(time_steps):
    u_opt = mpc.solve(x, target_state)
    x = x + np.array([0.1*(u_opt[0]-x[0]), 0.05*(u_opt[1]-x[1])])
    history.append({"time": t, "Temperature": x[0], "Pressure": x[1],
                    "HeaterPower": u_opt[0], "ValveOpening": u_opt[1]})

    axs[0].cla()
    axs[1].cla()
    axs[0].plot([h["time"] for h in history], [h["Temperature"] for h in history], label="Temperature")
    axs[0].plot([h["time"] for h in history], [h["Pressure"] for h in history], label="Pressure")
    axs[0].legend()
    axs[0].set_ylabel("Value")
    axs[0].set_title("States over Time")

    axs[1].plot([h["time"] for h in history], [h["HeaterPower"] for h in history], label="HeaterPower")
    axs[1].plot([h["time"] for h in history], [h["ValveOpening"] for h in history], label="ValveOpening")
    axs[1].legend()
    axs[1].set_ylabel("Control Value")
    axs[1].set_xlabel("Time Step")
    axs[1].set_title("Control Actions over Time")

    plt.pause(0.5)

plt.ioff()
