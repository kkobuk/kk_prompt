# 2️⃣ Backend: OWL 확장

### **ontology.py**

```python
from owlready2 import get_ontology, Thing, ObjectProperty, DataProperty, onto_path

onto_path.append("ontology")

def load_ontology(file_path):
    return get_ontology(f"{file_path}").load()

def save_ontology(onto, file_path="ontology/sample.owl"):
    onto.save(file=file_path)

# --- 클래스 ---
def get_classes(onto):
    return [cls.name for cls in onto.classes()]

def add_class(onto, class_name):
    with onto:
        NewClass = type(class_name, (Thing,), {})
    save_ontology(onto)
    return {"added": class_name}

# --- 관계 ---
def get_object_properties(onto):
    return [prop.name for prop in onto.object_properties()]

def add_object_property(onto, prop_name, domain, range_):
    with onto:
        NewProp = types.new_class(prop_name, (ObjectProperty,))
        NewProp.domain = [onto[domain]]
        NewProp.range = [onto[range_]]
    save_ontology(onto)
    return {"added": prop_name}

# --- 인스턴스 ---
def get_instances(onto):
    instances = []
    for cls in onto.classes():
        for inst in cls.instances():
            instances.append({"name": inst.name, "class": cls.name})
    return instances

def add_instance(onto, instance_name, class_name):
    cls = onto[class_name]
    instance = cls(instance_name)
    save_ontology(onto)
    return {"added": instance_name}
```

```javascript
import React, { useState, useEffect } from "react";
import { getObjectProperties, addObjectProperty, getClasses } from "../api/api";

const ObjectPropertyEditor = () => {
  const [properties, setProperties] = useState([]);
  const [classes, setClasses] = useState([]);
  const [newProp, setNewProp] = useState({ name: "", domain: "", range: "" });

  useEffect(() => {
    fetchProperties();
    fetchClasses();
  }, []);

  const fetchProperties = async () => {
    const data = await getObjectProperties();
    setProperties(data);
  };

  const fetchClasses = async () => {
    const data = await getClasses();
    setClasses(data);
  };

  const handleAddProperty = async () => {
    if (!newProp.name || !newProp.domain || !newProp.range) return;
    await addObjectProperty(newProp.name, newProp.domain, newProp.range);
    setNewProp({ name: "", domain: "", range: "" });
    fetchProperties();
  };

  return (
    <div>
      <h2>Object Properties</h2>
      <ul>{properties.map((p) => <li key={p}>{p}</li>)}</ul>
      <input placeholder="Name" value={newProp.name} onChange={e => setNewProp({...newProp, name: e.target.value})} />
      <select value={newProp.domain} onChange={e => setNewProp({...newProp, domain: e.target.value})}>
        <option value="">Domain</option>
        {classes.map(c => <option key={c} value={c}>{c}</option>)}
      </select>
      <select value={newProp.range} onChange={e => setNewProp({...newProp, range: e.target.value})}>
        <option value="">Range</option>
        {classes.map(c => <option key={c} value={c}>{c}</option>)}
      </select>
      <button onClick={handleAddProperty}>Add Property</button>
    </div>
  );
};

export default ObjectPropertyEditor;
```

```python id="backend-rename"
from fastapi import Body
from app.ontology import onto, save_ontology

@app.post("/rename_class")

    cls = onto[oldName]
    if cls is None:
        return {"error": "Class not found"}
    cls.name = newName
    save_ontology(onto)
    return {"status": f"{oldName} renamed to {newName}"}

@app.post("/rename_object_property")
def rename_object_property(oldName: str = Body(...), newName: str = Body(...)):
    prop = onto[oldName]
    if prop is None:
        return {"error": "Property not found"}
    prop.name = newName
    save_ontology(onto)
    return {"status": f"{oldName} renamed to {newName}"}
```

### ontology.py 확장

```python
from owlready2 import Thing, ObjectProperty, DataProperty, onto_path
onto_path.append("ontology")

# 인스턴스
def get_instances(onto):
    instances = []
    for cls in onto.classes():
        for inst in cls.instances():
            instances.append({"name": inst.name, "class": cls.name})
    return instances

def add_instance(onto, instance_name, class_name):
    cls = onto[class_name]
    instance = cls(instance_name)
    save_ontology(onto)
    return {"added": instance_name}

# DataProperty 생성
def add_data_property(onto, name, domain, data_type="str"):
    with onto:
        NewProp = type(name, (DataProperty,), {})
        NewProp.domain = [onto[domain]]
        # DataType 지정
        if data_type == "int": NewProp.range = [int]
        elif data_type == "float": NewProp.range = [float]
        else: NewProp.range = [str]
    save_ontology(onto)
    return {"added": name}
```

```javascript
    // 노드 클릭 → 이름 변경
    vowlGraph.on("nodeClicked", async (nodeData) => {
      const newName = prompt("Rename class/instance:", nodeData.id);
      if (!newName) return;

      if(nodeData.type === "instance"){
        await axios.post("http://localhost:8000/rename_instance", { oldName: nodeData.id, newName });
      } else {
        await axios.post("http://localhost:8000/rename_class", { oldName: nodeData.id, newName });
      }
      refreshGraph(vowlGraph);
    });

    // 엣지 클릭 → 이름 변경
    vowlGraph.on("edgeClicked", async (edgeData) => {
      const newName = prompt("Rename relation:", edgeData.id);
      if (!newName) return;

      if(edgeData.type === "objectProperty"){
        await axios.post("http://localhost:8000/rename_object_property", { oldName: edgeData.id, newName });
      } else if(edgeData.type === "dataProperty"){
        await axios.post("http://localhost:8000/rename_data_property", { oldName: edgeData.id, newName });
      }
      refreshGraph(vowlGraph);
    });

    // 빈 공간 더블 클릭 → 새 클래스/인스턴스 선택
    graphContainer.addEventListener("dblclick", async (event) => {
      const choice = prompt("Create: 1) Class  2) Instance");
      if(choice==="1"){
        const className = prompt("New class name:");
        if(className) await axios.post("http://localhost:8000/classes", { name: className });
      } else if(choice==="2"){
        const instName = prompt("Instance name:");
        const className = prompt("Class name for instance:");
        if(instName && className) await axios.post("http://localhost:8000/instances", { name: instName, class_name: className });
      }
      refreshGraph(vowlGraph);
    });

    // Reasoning 버튼
    document.getElementById("reasonBtn").addEventListener("click", async () => {
      await axios.post("http://localhost:8000/reason");
      const owlRes = await axios.get("http://localhost:8000/ontology/json");
      const owlJson = owlRes.data;

      vowlGraph.importData(owlJson);
      vowlGraph.render();

      // 추론 강조
      vowlGraph.nodes().forEach(node => { if(node.inferred) node.style.fill="#FF4136" });
      vowlGraph.edges().forEach(edge => { if(edge.inferred) edge.style.stroke="#FF4136" });
    });
  };

  const refreshGraph = async (vowlGraph) => {
    const owlRes = await axios.get("http://localhost:8000/ontology/json");
    vowlGraph.importData(owlRes.data);
    vowlGraph.render();
  };

  return (
    <div>
      <button id="reasonBtn">Run Reasoner & Highlight</button>
      <div id="graph" style={{ width: "100%", height: "600px" }} />
    </div>
  );
};

export default WebVowlEditor;
```

```python
from owlready2 import get_ontology, Thing, ObjectProperty, DataProperty, onto_path
import types

onto_path.append("ontology")

def load_ontology(file_path="ontology/sample.owl"):
    return get_ontology(f"{file_path}").load()

def save_ontology(onto, file_path="ontology/sample.owl"):
    onto.save(file=file_path)

# 클래스
def get_classes(onto):
    return [cls.name for cls in onto.classes()]

def add_class(onto, class_name):
    with onto:
        NewClass = type(class_name, (Thing,), {})
    save_ontology(onto)
    return {"added": class_name}

# ObjectProperty
def get_object_properties(onto):
    return [prop.name for prop in onto.object_properties()]

def add_object_property(onto, name, domain, range_):
    with onto:
        NewProp = types.new_class(name, (ObjectProperty,))
        NewProp.domain = [onto[domain]]
        NewProp.range = [onto[range_]]
    save_ontology(onto)
    return {"added": name}

# DataProperty
def add_data_property(onto, name, domain, data_type="str"):
    with onto:
        NewProp = type(name, (DataProperty,), {})
        NewProp.domain = [onto[domain]]
        if data_type == "int": NewProp.range = [int]
        elif data_type == "float": NewProp.range = [float]
        else: NewProp.range = [str]
    save_ontology(onto)
    return {"added": name}

# 인스턴스
def get_instances(onto):
    instances = []
    for cls in onto.classes():
        for inst in cls.instances():
            instances.append({"name": inst.name, "class": cls.name})
    return instances

def add_instance(onto, instance_name, class_name):
    cls = onto[class_name]
    instance = cls(instance_name)
    save_ontology(onto)
    return {"added": instance_name}
```

# 3️⃣ ZIP 패키지 생성 스크립트

`create_full_package.py`:

```python
import os, zipfile

root = "rdf_owl_gui_full_package"
zip_name = "rdf_owl_gui_full_package.zip"

with zipfile.ZipFile(zip_name, 'w', zipfile.ZIP_DEFLATED) as zipf:
    for foldername, subfolders, filenames in os.walk(root):
        for filename in filenames:
            filepath = os.path.join(foldername, filename)
            arcname = os.path.relpath(filepath, root)
            zipf.write(filepath, arcname)

print(f"완전 실행 ZIP 생성 완료: {zip_name}")
```

> WebVOWL + D3 파일을 미리 `frontend/public/webvowl/`에 넣고 실행해야 합니다.

```bash id="qw7g5s"
# Markdown 파일 생성 (tutorial.md)
# 예: Markdown 안내서 내용 전체

# PDF로 변환
python -c "
import markdown2, pdfkit
with open('tutorial.md') as f:
    html = markdown2.markdown(f.read())
pdfkit.from_string(html, 'RDF_OWL_GUI_Tutorial.pdf')
"
```

* `Ctrl+Shift+V`로 Preview

# 📄 `tutorial.md` Markdown 템플릿

````markdown
# RDF/OWL GUI Editor Tutorial

## 1️⃣ 준비 단계
1. GitHub Release 또는 ZIP 파일 다운로드
2. 압축 해제
3. Backend 설치 및 실행
```bash
cd backend
python -m venv venv
source venv/bin/activate  # Windows: venv\Scripts\activate
pip install -r requirements.txt
uvicorn app.main:app --reload
````

4. Frontend 설치 및 실행

```bash
cd ../frontend
npm install
npm start
```

5. 브라우저 접속: [http://localhost:3000](http://localhost:3000)

---

## 2️⃣ GUI 기본 화면

* 좌측: 클래스 노드
* 중앙: 인스턴스 노드
* 우측: ObjectProperty / DataProperty 연결
* 상단 메뉴: **Add Class / Add Individual / Add ObjectProperty / Add DataProperty / Reasoner / Export OWL**
* ![그림1](/images/gui_initial.png)  <!-- GUI 초기 화면 캡처 -->

---

## 3️⃣ 클래스 추가

1. 메뉴 → **Add Class**
2. 이름 입력: `Publisher`
3. 확인 → 노드 생성

* ![그림2](/images/class_publisher.png)  <!-- Publisher 노드 생성 -->

---

## 4️⃣ ObjectProperty 연결

1. 메뉴 → **Add Object Property**
2. 이름: `publishes`
3. Domain: `Publisher`
4. Range: `Book`
5. 확인 → Edge 생성

* ![그림3](/images/objectproperty_edge.png)  <!-- Edge 연결 화면 -->

---

## 5️⃣ 인스턴스 추가

1. 메뉴 → **Add Individual**
2. 이름: `Bob`
3. Type: `Person`
4. 확인 → 노드 생성
5. DataProperty 설정: `age = 45`

* ![그림4](/images/individual_bob.png)  <!-- Bob 노드 및 속성 -->

---

## 6️⃣ Reasoning 실행

* 메뉴 → **Reasoner** 클릭
* 추론 결과:

  * 인스턴스 간 관계 강조 표시
  * 자동 생성된 Edge 확인
* ![그림5](/images/reasoning_result.png)  <!-- Reasoning 결과 -->

---

## 7️⃣ OWL 파일 내보내기

* 메뉴 → **Export OWL**
* 파일 저장 → `backend/ontology/` 덮어쓰기 가능
* 서버 재시작 후 새 OWL 로드
